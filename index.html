Setup Database Supabase

Salin query SQL di bawah ini dan jalankan di menu SQL Editor pada Dashboard Supabase Anda. 

DROP TABLE IF EXISTS initial_balances CASCADE;
DROP TABLE IF EXISTS transactions CASCADE;

CREATE TABLE initial_balances (
    id INT PRIMARY KEY DEFAULT 1,
    app_balance NUMERIC DEFAULT 0,
    cash_balance NUMERIC DEFAULT 0,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    CONSTRAINT single_row CHECK (id = 1)
);

CREATE TABLE transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    type VARCHAR(20) NOT NULL,
    food_price NUMERIC DEFAULT 0,
    income NUMERIC DEFAULT 0,
    expense_amount NUMERIC DEFAULT 0,
    source VARCHAR(20) DEFAULT 'CASH',
    notes TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

INSERT INTO initial_balances (id, app_balance, cash_balance)
VALUES (1, 0, 0)
ON CONFLICT (id) DO NOTHING;

ALTER TABLE initial_balances DISABLE ROW LEVEL SECURITY;
ALTER TABLE transactions DISABLE ROW LEVEL SECURITY;
